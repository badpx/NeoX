cmake_minimum_required(VERSION 3.16)
project(NeoXConsumer C)
if(NEOX_SOURCE)
  add_subdirectory("${NEOX_SOURCE}" neox)
else()
  find_package(NeoX 0.1 CONFIG REQUIRED COMPONENTS Core ${NEOX_CONSUMER_COMPONENTS})
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE NeoX::core)
if(TARGET NeoX::text)
  target_link_libraries(consumer PRIVATE NeoX::text)
endif()
if(TARGET NeoX::present)
  target_link_libraries(consumer PRIVATE NeoX::present)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(consumer PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
set_tests_properties(consumer PROPERTIES ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
